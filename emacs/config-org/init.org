#+begin_src emacs-lisp
;; Load core modules in order
;; 1. core-bootstrap: paths, GC optimization, logging
(require 'core-bootstrap)

;; Initialize paths based on init.el location
;; init.el is at: emacs/site-lisp/config/init.el
;; We need to go up two levels to get emacs/ directory
;; config/ -> site-lisp/ -> emacs/
(let* ((config-dir (file-name-as-directory mcg-init-file-dir))
       ;; Remove trailing slash, then get parent directory
       (site-lisp-dir (file-name-directory (directory-file-name config-dir)))
       (emacs-dir (file-name-directory (directory-file-name site-lisp-dir))))
  (mcg-init-paths emacs-dir))

;; Setup load-path for all modules and extensions
(mcg-setup-load-path)

;; Setup GC optimization
(mcg-setup-gc-optimization)
#+end_src

** Setup Lazy Command Entries

#+begin_src emacs-lisp
(defun mcg--idle-module-command-entries-enabled-p (category module)
  "Return non-nil when MODULE in CATEGORY should expose lazy command entries."
  (and (mcg-module-enabled-p category module)
       (eq (mcg-module-load-strategy category module) 'idle)))

(defun mcg--bind-idle-module-command-entries (category module source bindings)
  "Bind lazy command entry BINDINGS for idle MODULE in CATEGORY.
BINDINGS is an alist of (KEY . COMMAND)."
  (when (mcg--idle-module-command-entries-enabled-p category module)
    (dolist (binding bindings)
      (mcg-bind-lazy-module-key
       (car binding) (cdr binding) source category module))))

(defun mcg-setup-idle-module-command-entries ()
  "Expose high-value commands from idle modules immediately."
  (when (mcg--idle-module-command-entries-enabled-p :tools 'compile)
    (define-prefix-command 'mcg-compile-map)
    (mcg-bind-key "C-c c" 'mcg-compile-map "tools/compile")
    (mcg--bind-idle-module-command-entries
     :tools 'compile "tools/compile"
     '(("C-c c c" . mcg-compile)
       ("C-c c r" . mcg-recompile)
       ("C-c c b" . mcg-compile-build)
       ("C-c c t" . mcg-compile-test)
       ("C-c c R" . mcg-compile-run)
       ("M-g e" . mcg-compile-errors))))

  (mcg--bind-idle-module-command-entries

** Load Immediate Modules

#+begin_src emacs-lisp
(defun mcg-load-immediate-modules ()
  "Load all modules declared with `:load immediate'."
  (dolist (mod-key (mcg-list-modules-by-load-strategy 'immediate))
    (mcg-load-module (car mod-key) (cdr mod-key))))
#+end_src

** Setup Idle Loading

#+begin_src emacs-lisp
(defun mcg-setup-idle-modules ()
  "Set up idle loading for modules declared with `:load idle'."
  (dolist (mod-key (reverse (mcg-list-modules-by-load-strategy 'idle)))
    (mcg-add-idle-load-module (car mod-key) (cdr mod-key)))
  (mcg-setup-idle-loading))
#+end_src

** Setup Deferred Loading

#+begin_src emacs-lisp
(defun mcg-setup-deferred-modules ()
  "Set up trigger-based loading for modules declared with `:load deferred'."
  (dolist (mod-key (mcg-list-modules-by-load-strategy 'deferred))
    (mcg-defer-module (car mod-key) (cdr mod-key))))

** Load Private Configuration

#+begin_src emacs-lisp
(defun mcg-load-private-file (filename profile-name)
  "Load private FILENAME under `mcg-private-dir' using PROFILE-NAME."
  (let ((private-file (expand-file-name filename mcg-private-dir)))
    (when (file-exists-p private-file)
      (condition-case err
          (progn
            (mcg-profile profile-name
              (load private-file nil t))
            (mcg-log "Loaded %s" profile-name))
        (error
         (mcg-log-error "Failed to load %s: %s"
                        profile-name
                        (error-message-string err)))))))

(defun mcg-load-private-config ()
  "Load private config."
  (mcg-load-private-file "config.el" "private/config"))
#+end_src

** Startup Statistics

#+begin_src emacs-lisp
(defun mcg-display-startup-stats ()
  "Display startup statistics after Emacs is ready."
  (let ((loaded-count (length mcg-loaded-modules))
        (deferred-count (length mcg-deferred-modules))
        (idle-count (length mcg-idle-load-modules))
        (error-count (length mcg-module-load-errors)))
    (mcg-log "Startup complete: %d loaded, %d deferred, %d idle, %d errors"
             loaded-count deferred-count idle-count error-count)
    ;; Show errors if any
    (when (> error-count 0)
      (mcg-log-warning "Module load errors occurred. Run M-x mcg-list-modules to see details."))))
#+end_src

** Main Initialization Sequence

#+begin_src emacs-lisp
(defun mcg-init ()
  "Main initialization function.
Called to set up the complete MCG Emacs configuration."
  ;; Expose selected idle-module commands without paying their startup cost.
  (mcg-setup-idle-module-command-entries)

  ;; Load immediate modules (UI, editor, tools, org)
  (mcg-load-immediate-modules)

  ;; Setup idle loading for completion and enhance modules
  (mcg-setup-idle-modules)

  ;; Setup deferred loading for writing and language modules
  (mcg-setup-deferred-modules)

  ;; Load private configuration last
  (mcg-load-private-config)

  ;; Display startup statistics
  (add-hook 'emacs-startup-hook #'mcg-display-startup-stats 95))

;; Run initialization
(mcg-init)
#+end_src
