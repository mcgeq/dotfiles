* Shared Module Command Bindings

Idle entry points and initialized modules consume the same declarations.
Keys expressed as symbols resolve private options at binding time.

(defun mcg-bind-module-commands (category module &optional lazy)
  "Install shared bindings for MODULE in CATEGORY.
When LAZY is non-nil, commands initialize their module before execution."
  (let ((entry (cl-find-if (lambda (row)
                             (and (eq (nth 0 row) category)
                                  (eq (nth 1 row) module)))
                           mcg-module-command-bindings)))
    (when entry
      (when (and (eq category :tools) (eq module 'compile))
        (mcg-ensure-prefix-map "C-c c" 'mcg-compile-map "tools/compile"))
      (when (and (eq category :tools) (memq module '(blink-search color-rg)))
        (mcg-ensure-prefix-map "C-c s" 'mcg-search-map "completion/consult"))
      (dolist (binding (nthcdr 3 entry))
        (let ((key (if (symbolp (car binding))
                       (symbol-value (car binding))
                     (car binding)))
              (command (cdr binding))
              (source (nth 2 entry)))
          (if lazy
              (mcg-bind-lazy-module-key key command source category module)
            (mcg-bind-key key command source)))))))
