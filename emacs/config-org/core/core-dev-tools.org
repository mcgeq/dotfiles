* Development Tool Selection

Shared tool selection for compile, format, doctor, and language commands.
Requiring this module does not activate language modes or install their hooks.
Existing Python/format/Zig helper names and options remain available for compatibility.

** Python Options and Commands

#+begin_src emacs-lisp
(defcustom mcg-python-test-command "python -m pytest"
  "Command used to run Python tests without a project runner."
  :type 'string
  :group 'mcg)

(defcustom mcg-python-ruff-command "ruff"
  "Command used to run Ruff."
  :type 'string
  :group 'mcg)

(defcustom mcg-python-detect-project-tools t
  "Whether Python commands prefer available uv, Poetry, or PDM runners."
  :type 'boolean
  :group 'mcg)

(defconst mcg-python-ruff-format-quote-style "format.quote-style = 'single'"
  "Ruff format config that preserves Python strings as single quotes.")

(defconst mcg-python-ruff-format-config-file-name "ruff-format-single.toml"
  "Filename for the generated Ruff formatter config.")

(defun mcg-python--project-root ()
  "Return the current Python project root or `default-directory'."
  (mcg-project-root-or-default))

(defun mcg-python--project-file-p (root file)
  "Return non-nil when ROOT contains FILE."
  (file-exists-p (expand-file-name file root)))

(defun mcg-python--project-tool (&optional root)
  "Return an available Python project runner for ROOT, or nil."
  (let ((root (or root (mcg-python--project-root))))
    (when mcg-python-detect-project-tools
      (cl-loop for (file . tool) in '(("uv.lock" . "uv")
                                      ("poetry.lock" . "poetry")
                                      ("pdm.lock" . "pdm"))
               when (and (mcg-python--project-file-p root file)
                         (executable-find tool))
               return tool))))

(defun mcg-python--project-tool-command (tool args &optional root)
  "Return project-aware command parts for TOOL with ARGS in ROOT."
  (let ((runner (mcg-python--project-tool root)))
    (append (when runner (list runner "run")) (cons tool args))))

(defun mcg-python--command-string (parts)
  "Return shell command string from command PARTS."
  (string-join parts " "))

(defun mcg-python--shell-command-string (parts)
  "Return a shell-quoted command string from command PARTS."
  (string-join (mapcar #'shell-quote-argument parts) " "))

(defun mcg-python--pytest-command (&optional root)
  "Return the best pytest command for ROOT."
  (if (mcg-python--project-tool root)
      (mcg-python--command-string
       (mcg-python--project-tool-command "pytest" nil root))
    mcg-python-test-command))

(defun mcg-python--pyproject-has-ruff-config-p (file)
  "Return non-nil when FILE contains a Ruff pyproject section."
  (and (file-exists-p file)
       (with-temp-buffer
         (insert-file-contents file)
         (goto-char (point-min))
         (re-search-forward "^\\[tool\\.ruff\\(?:\\.[^]]+\\)?\\]$" nil t))))

(defun mcg-python--ruff-project-config-p (&optional root)
  "Return non-nil when ROOT declares Ruff configuration."
  (let ((root (or root (mcg-python--project-root))))
    (or (mcg-python--project-file-p root ".ruff.toml")
        (mcg-python--project-file-p root "ruff.toml")
        (mcg-python--pyproject-has-ruff-config-p
         (expand-file-name "pyproject.toml" root)))))

(defun mcg-python--ruff-format-config-file ()
  "Return a Ruff config file path that forces single-quote formatting."
  (let* ((dir (or mcg-cache-dir temporary-file-directory))
         (file (expand-file-name mcg-python-ruff-format-config-file-name dir))
         (contents "[format]\nquote-style = \"single\"\n"))
    (unless (file-directory-p dir) (make-directory dir t))
    (unless (and (file-exists-p file)
                 (with-temp-buffer
                   (insert-file-contents file)
                   (equal (buffer-string) contents)))
      (with-temp-file file (insert contents)))
    file))

(defun mcg-python--ruff-format-args (&optional root)
  "Return Ruff format args for ROOT without overriding project config."
  (if (mcg-python--ruff-project-config-p root)
      '("format")
    (list "format" "--config" (mcg-python--ruff-format-config-file))))

(defun mcg-python-format-candidates (&optional root)
  "Return project-aware Python formatter command candidates for ROOT."
  (let* ((root (or root (mcg-python--project-root)))
         (args (mcg-python--ruff-format-args root)))
    (append (when (mcg-python--project-tool root)
              (list (mcg-python--project-tool-command "ruff" args root)))
            (list (cons mcg-python-ruff-command args)
                  (list "black" "--skip-string-normalization")))))

(defun mcg-python--ruff-format-command (&optional root)
  "Return Ruff format command parts for ROOT."
  (mcg-python--project-tool-command "ruff" (mcg-python--ruff-format-args root) root))

** Web Project Inspection

#+begin_src emacs-lisp
(defun mcg-format--project-root ()
  "Return current project root for formatter detection."
  (mcg-project-root-or-default))

(defun mcg-format--root-file (root file)
  "Return FILE expanded under ROOT."
  (expand-file-name file root))

(defun mcg-format--root-has-file-p (root file)
  "Return non-nil when ROOT contains FILE."
  (file-exists-p (mcg-format--root-file root file)))

(defun mcg-format--root-has-any-file-p (root files)
  "Return non-nil when ROOT contains any file in FILES."
  (cl-some (lambda (file) (mcg-format--root-has-file-p root file)) files))

(defun mcg-format--read-package-json (root)
  "Read package.json from ROOT as an alist, or nil when unavailable."
  (let ((file (mcg-format--root-file root "package.json")))
    (when (file-exists-p file)
      (condition-case nil
          (let ((json-object-type 'alist)
                (json-array-type 'list)
                (json-key-type 'symbol))
            (json-read-file file))
        (error nil)))))

(defun mcg-format--package-section (package section)
  "Return SECTION from PACKAGE alist."
  (cdr (assoc section package)))

(defun mcg-format--package-has-key-p (package section key)
  "Return non-nil when PACKAGE SECTION contains KEY."
  (assoc-string key (mcg-format--package-section package section) t))

(defun mcg-format--package-has-any-key-p (package sections key)
  "Return non-nil when PACKAGE has KEY in any of SECTIONS."
  (cl-some (lambda (section)
             (mcg-format--package-has-key-p package section key)) sections))

(defun mcg-format--file-contains-p (file regexp)
  "Return non-nil when FILE contains REGEXP."
  (when (file-exists-p file)
    (with-temp-buffer
      (insert-file-contents file)
      (goto-char (point-min))
      (re-search-forward regexp nil t))))

(defun mcg-format--root-file-contains-any-p (root files regexp)
  "Return non-nil when any file in FILES under ROOT contains REGEXP."
  (cl-some (lambda (file)
             (mcg-format--file-contains-p (mcg-format--root-file root file) regexp))
           files))

(defun mcg-format--web-eslint-config-files ()
  "Return known ESLint config file names."
  '("eslint.config.js" "eslint.config.mjs" "eslint.config.cjs" "eslint.config.ts"
    ".eslintrc" ".eslintrc.js" ".eslintrc.cjs" ".eslintrc.json"
    ".eslintrc.yaml" ".eslintrc.yml"))

(defun mcg-format--web-prettier-config-files ()
  "Return known Prettier config file names."
  '(".prettierrc" ".prettierrc.json" ".prettierrc.json5" ".prettierrc.yaml"
    ".prettierrc.yml" ".prettierrc.js" ".prettierrc.cjs" ".prettierrc.mjs"
    "prettier.config.js" "prettier.config.cjs" "prettier.config.mjs"))

(defun mcg-format--web-project-has-antfu-eslint-p (&optional root)
  "Return non-nil when ROOT declares Antfu ESLint configuration."
  (let* ((root (or root (mcg-format--project-root)))
         (package (mcg-format--read-package-json root)))
    (or (mcg-format--package-has-any-key-p
         package '(dependencies devDependencies optionalDependencies peerDependencies)
         "@antfu/eslint-config")
        (mcg-format--root-file-contains-any-p
         root (mcg-format--web-eslint-config-files) "@antfu/eslint-config"))))

(defun mcg-format--web-project-has-eslint-p (&optional root)
  "Return non-nil when ROOT declares ESLint formatting/linting."
  (let* ((root (or root (mcg-format--project-root)))
         (package (mcg-format--read-package-json root)))
    (or (mcg-format--web-project-has-antfu-eslint-p root)
        (mcg-format--root-has-any-file-p root (mcg-format--web-eslint-config-files))
        (assoc 'eslintConfig package)
        (mcg-format--package-has-any-key-p
         package '(dependencies devDependencies optionalDependencies peerDependencies)
         "eslint"))))

(defun mcg-format--web-project-has-prettier-p (&optional root)
  "Return non-nil when ROOT explicitly declares Prettier."
  (let* ((root (or root (mcg-format--project-root)))
         (package (mcg-format--read-package-json root)))
    (or (mcg-format--root-has-any-file-p root (mcg-format--web-prettier-config-files))
        (assoc 'prettier package)
        (mcg-format--package-has-any-key-p
         package '(dependencies devDependencies optionalDependencies peerDependencies)
         "prettier"))))

(defun mcg-format--node-tool-command (root tool args)
  "Return command candidate for TOOL with ARGS in ROOT."
  (cond
   ((or (mcg-format--root-has-file-p root "pnpm-lock.yaml")
        (mcg-format--root-has-file-p root "pnpm-workspace.yaml"))
    (append (list "pnpm" "exec" tool) args))
   ((or (mcg-format--root-has-file-p root "bun.lock")
        (mcg-format--root-has-file-p root "bun.lockb"))
    (append (list "bun" "x" tool) args))
   ((mcg-format--root-has-file-p root "package-lock.json")
    (append (list "npm" "exec" "--" tool) args))
   (t (cons tool args))))

(defun mcg-format--web-external-candidates ()
  "Return project-aware Web formatter candidates, preferring ESLint."
  (let ((root (mcg-format--project-root)))
    (cond
     ((mcg-format--web-project-has-eslint-p root)
      (list (mcg-format--node-tool-command root "eslint" '("--fix"))))
     ((mcg-format--web-project-has-prettier-p root)
      (list (mcg-format--node-tool-command root "prettier" '("--write")))))))
