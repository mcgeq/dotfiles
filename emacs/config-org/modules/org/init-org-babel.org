;;; ============================================================

(defcustom mcg-org-babel-confirm-evaluate t
  "Whether Org should ask before evaluating Babel code blocks."
  :type 'boolean
  :group 'mcg-org)

(defcustom mcg-org-export-babel-evaluate nil
  "Whether Org export should evaluate Babel code blocks."
  :type 'boolean
  :group 'mcg-org)

(defcustom mcg-org-babel-load-languages
  ;; Conservative default languages: keep the common literate config and
  ;; day-to-day scripting surface small. Extended Babel languages such as C,
  ;; dot, plantuml, sql, and Jupyter can be enabled from private config when
  ;; needed.
  '((emacs-lisp . t)
    (shell . t)
    (python . t)
    (js . t)
    (css . t)
    (org . t))
  "Languages enabled for Org Babel.
Zig is controlled separately by `mcg-org-babel-enable-zig'.
Jupyter is controlled separately by `mcg-org-babel-enable-jupyter'."
  :type '(repeat (cons symbol boolean))
  :group 'mcg-org)

(defcustom mcg-org-babel-enable-jupyter nil
  "Whether to enable Org Babel support for Jupyter kernels."
  :type 'boolean
  :group 'mcg-org)

(defcustom mcg-org-babel-jupyter-default-header-args
  '((:async . "yes")
    (:session . "py")
    (:kernel . "python3"))
  "Default header arguments for Jupyter Org Babel blocks."
  :type 'sexp
  :group 'mcg-org)

(defcustom mcg-org-babel-enable-zig t
  "Whether to enable the custom Zig Org Babel executor."
  :type 'boolean
  :group 'mcg-org)

(defcustom mcg-org-babel-zig-default-header-args
  '((:results . "output")
    (:exports . "both"))
  "Default header arguments for Zig Org Babel blocks."
  :type 'sexp
  :group 'mcg-org)

(defcustom mcg-org-babel-src-block-languages
  '("emacs-lisp" "python" "shell" "zig" "js" "css" "org")
  "Language candidates offered by `mcg/org-insert-src-block'.
Jupyter adds `jupyter-python' when enabled and available."
  :type '(repeat string)
  :group 'mcg-org)

(defvar mcg-org-babel--jupyter-missing-warned nil
  "Non-nil means the missing Jupyter CLI warning has already been shown.")

(defun mcg-org-babel--jupyter-available-p ()
  "Return non-nil when the Jupyter CLI is available."
  (executable-find "jupyter"))

(defun mcg-org-babel--maybe-enable-jupyter-defaults ()
  "Apply Jupyter Babel defaults when configured and available."
  (when (and mcg-org-babel-enable-jupyter
             (mcg-org-babel--jupyter-available-p))
    (setq org-babel-default-header-args:jupyter-python
          mcg-org-babel-jupyter-default-header-args)
    (add-to-list 'mcg-org-babel-src-block-languages "jupyter-python")))

(defun mcg-org-babel--effective-load-languages ()
  "Return the Org Babel languages to load for this session."
  (let ((languages (copy-sequence mcg-org-babel-load-languages)))
    (when mcg-org-babel-enable-jupyter
      (if (mcg-org-babel--jupyter-available-p)
          ;; emacs-jupyter expects the jupyter language to be loaded last.
          (setq languages (append languages '((jupyter . t))))
        (unless mcg-org-babel--jupyter-missing-warned
          (setq mcg-org-babel--jupyter-missing-warned t)
          (mcg-log-warning "jupyter executable not found; Jupyter Babel disabled"))))
    languages))

(defun mcg-org-babel-setup-basic ()
  "Setup Org Babel basic configuration."
  (setq org-confirm-babel-evaluate mcg-org-babel-confirm-evaluate
        org-src-fontify-natively t
        org-src-tab-acts-natively t
        org-src-preserve-indentation t
        org-edit-src-content-indentation 0)
  (setq org-src-window-setup 'current-window)
  (setq org-export-use-babel t
        org-export-babel-evaluate mcg-org-export-babel-evaluate)
  (mcg-log "Configured org-babel basic settings"))

** Zig Language Support

#+begin_src emacs-lisp
;;; Zig Language Support

(defvar org-babel-default-header-args:zig '()
  "Default arguments for Zig code blocks.")

(defun org-babel-execute:zig (body params)
  "Execute Zig code block.
BODY is the code content, PARAMS are the parameters."
  (let* ((tmp-src-file (org-babel-temp-file "zig-src-" ".zig"))
         (tmp-bin-file (org-babel-temp-file "zig-bin-"))
         (cmdline (cdr (assq :cmdline params)))
         (flags (cdr (assq :flags params)))
         (full-body (org-babel-expand-body:generic body params)))

    (unless (executable-find "zig")
      (error "Zig compiler not found"))

    (with-temp-file tmp-src-file
      (insert full-body))

    (let ((compile-command
           (format "zig build-exe %s %s -femit-bin=%s"
                   (or flags "")
                   (org-babel-process-file-name tmp-src-file)
                   (org-babel-process-file-name tmp-bin-file)))
          (run-command
           (format "%s %s"
                   (org-babel-process-file-name tmp-bin-file)
                   (or cmdline ""))))

      (let ((compile-result (org-babel-eval compile-command "")))
        (if (string-match-p "error" compile-result)
            (concat "Compilation failed:\n" compile-result)
          (org-babel-eval run-command ""))))))

(defun mcg-org-babel-setup-zig ()
  "Setup Zig language support."
  (with-eval-after-load 'org
    (when mcg-org-babel-enable-zig
      (add-to-list 'org-babel-load-languages '(zig . t))
      (setq org-babel-default-header-args:zig
            mcg-org-babel-zig-default-header-args)
      (mcg-log "Configured org-babel Zig support"))))
#+end_src

** Helper Functions

#+begin_src emacs-lisp
;;; Helper Functions

(defun mcg/org-babel-execute-buffer ()
  "Execute all code blocks in current buffer."
  (interactive)
  (org-babel-execute-buffer)
  (message "Executed all code blocks in buffer"))

(defun mcg/org-babel-tangle-and-compile ()
  "Tangle current file."
  (interactive)
  (org-babel-tangle)
  (message "Tangled file"))

(defun mcg/org-babel-remove-all-results ()
  "Remove all code block results in current buffer."
  (interactive)
  (org-babel-remove-result-one-or-many t)
  (message "Removed all results"))

(defun mcg/org-insert-src-block (lang)
  "Insert a code block for the specified language LANG."
  (interactive
   (list (completing-read "Language: "
                          mcg-org-babel-src-block-languages
                          nil nil)))
  (insert (format "#+begin_src %s\n\n#+end_src" lang))
  (forward-line -1))
#+end_src

;;; Keybindings

(defun mcg-org-babel-setup-keybindings ()
  "Setup Org Babel keybindings."
  (with-eval-after-load 'org
    (define-key org-mode-map (kbd "C-c C-v C-b") #'mcg/org-babel-execute-buffer)
    (define-key org-mode-map (kbd "C-c C-v C-r") #'mcg/org-babel-remove-all-results)
    (define-key org-mode-map (kbd "C-c C-v i") #'mcg/org-insert-src-block))
  (mcg-log "Configured org-babel keybindings"))
