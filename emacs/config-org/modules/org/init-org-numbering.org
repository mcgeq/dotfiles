** Configuration

#+begin_src emacs-lisp
;;; ============================================================
;;; Configuration
;;; ============================================================

(defcustom mcg-org-enable-org-numbering nil
  "Whether to enable org-numbering in Org buffers."
  :type 'boolean
  :group 'mcg-org)

(defun mcg-org-numbering-enable ()
  "Enable org-numbering in the current buffer."
  (org-numbering-mode 1))

(defun mcg--setup-org-numbering ()
  "Set up org-numbering."
  (with-eval-after-load 'org
    (when (and mcg-org-enable-org-numbering
               (mcg-require-extension "org-numbering" t))
      (add-hook 'org-mode-hook #'mcg-org-numbering-enable)
      (mcg-log "Org numbering configured"))))
#+end_src

** Initialize

#+begin_src emacs-lisp
(mcg--setup-org-numbering)
#+end_src
