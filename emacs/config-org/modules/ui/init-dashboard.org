(defcustom mcg-dashboard-enabled t
  "Whether to enable dashboard."
  :type 'boolean

  '((recents . 5)

  "Icon backend used by dashboard.

** Setup

#+begin_src emacs-lisp
(defun mcg-dashboard-icons-enabled-p ()
  "Return non-nil when dashboard icons should be enabled."
  (and mcg-dashboard-icon-type
       (require mcg-dashboard-icon-type nil t)))

(defun mcg--setup-dashboard ()
  "Setup dashboard."
  (if (and mcg-dashboard-enabled
           (mcg-require-extension "dashboard" t))
      (progn
        (setq dashboard-banner-logo-title mcg-dashboard-banner-title)
        (setq dashboard-startup-banner mcg-dashboard-startup-banner)
        (setq dashboard-banner-ascii mcg-dashboard-banner-ascii)
        (setq dashboard-footer-messages mcg-dashboard-footer-messages)
        (setq dashboard-items mcg-dashboard-items)
        (setq dashboard-projects-backend 'project-el)
        (setq dashboard-center-content t)
        (setq dashboard-show-shortcuts nil)
        (setq dashboard-set-heading-icons (mcg-dashboard-icons-enabled-p))
        (setq dashboard-set-file-icons dashboard-set-heading-icons)
        (setq dashboard-icon-type
              (when dashboard-set-heading-icons mcg-dashboard-icon-type))
        (when (eq dashboard-icon-type 'nerd-icons)
          (setq dashboard-heading-icons mcg-dashboard-nerd-icons-heading-icons))
        (when mcg-dashboard-startup
          (setq initial-buffer-choice #'dashboard-open)
          (dashboard-setup-startup-hook))
        (mcg-log "dashboard configured"))
    (mcg-log-warning "dashboard not available or disabled")))

(defun mcg/dashboard-open ()
  "Open dashboard."
  (interactive)
  (if (fboundp 'dashboard-open)
      (dashboard-open)
    (message "dashboard is not available")))
#+end_src
