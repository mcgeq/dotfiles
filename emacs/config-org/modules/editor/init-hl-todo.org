** Setup

#+begin_src emacs-lisp
(defun mcg--setup-hl-todo ()
  "Setup global hl-todo keyword highlighting."
  (when (and (mcg-require-extension "compat" t)
             (mcg-require-extension "cond-let" t)
             (mcg-require-extension "hl-todo" t))
    (setq hl-todo-keyword-faces mcg-hl-todo-keyword-faces)
    (setq hl-todo-include-modes '(prog-mode text-mode conf-mode))
    (setq hl-todo-exclude-modes '(org-mode))
    (global-hl-todo-mode 1)
    (mcg-log "hl-todo configured")))

(defun mcg-hl-todo-grep-regexp ()
  "Return grep-friendly regexp for configured hl-todo keywords."
  (with-temp-buffer
    (hl-todo--regexp t)))

(defun mcg/hl-todo-search-project ()
  "Search TODO-style keywords in the current project."
  (interactive)
  (hl-todo-rgrep (mcg-hl-todo-grep-regexp)
                 "*"
                 (mcg-project-root-or-default)
                 nil))
#+end_src

** Initialize

#+begin_src emacs-lisp

(mcg-bind-key "C-c s T" #'mcg/hl-todo-search-project "editor/hl-todo")
#+end_src
