#+begin_src emacs-lisp
;;; ============================================================
;;; zig-mode / zig-ts-mode Configuration
;;; ============================================================

(defun mcg-zig--treesit-available-p ()
  "Return non-nil when Zig tree-sitter grammar is available."
  (and (fboundp 'treesit-language-available-p)
       (treesit-language-available-p 'zig)))

(defun mcg-zig-deferred-mode ()
  "Select Zig major mode after the zig module is deferred-loaded."
  (if (and (fboundp 'zig-ts-mode)
           (mcg-zig--treesit-available-p))
      (zig-ts-mode)
    (zig-mode)))

(defun mcg-zig-mode-setup ()
  "Setup function for zig-mode and zig-ts-mode."
  ;; Indentation
  (setq-local indent-tabs-mode nil)
  (setq-local tab-width 4)

  ;; Enable subword mode for camelCase navigation
  (subword-mode 1)

  (add-hook 'kill-buffer-hook #'mcg-zig--cancel-lsp-attach-timer nil t)
  (mcg-zig-schedule-lsp)

  (mcg-log "Zig mode setup complete"))

(defun mcg-zig--zls-available-p ()
  "Return non-nil when the Zig language server executable is available."
  (when (eq mcg-zig--zls-executable 'unknown)
    (setq mcg-zig--zls-executable (executable-find "zls")))
  mcg-zig--zls-executable)

(defun mcg-zig--warn-missing-zls ()
  "Warn once when `zls' is unavailable."
  (when (and mcg-zig-warn-missing-zls
             (not mcg-zig--missing-zls-warned))
    (setq mcg-zig--missing-zls-warned t)
    (mcg-log-warning "zls executable not found; Zig LSP disabled")))

(defun mcg-zig--buffer-p (&optional buffer)
  "Return non-nil when BUFFER is a Zig editing buffer."
  (with-current-buffer (or buffer (current-buffer))
    (derived-mode-p 'zig-mode 'zig-ts-mode)))

(defun mcg-zig--cancel-lsp-attach-timer ()
  "Cancel any pending deferred LSP startup for the current Zig buffer."
  (when (timerp mcg-zig--lsp-attach-timer)
    (cancel-timer mcg-zig--lsp-attach-timer))
  (setq mcg-zig--lsp-attach-timer nil))

(defun mcg-zig--enable-lsp-in-buffer (buffer)
  "Enable `lsp-bridge' in BUFFER when Zig LSP prerequisites are ready."
  (when (buffer-live-p buffer)
    (with-current-buffer buffer
      (setq mcg-zig--lsp-attach-timer nil)
      (when (and mcg-zig-enable-lsp
                 (mcg-zig--buffer-p)
                 (mcg-zig--zls-available-p)
                 (fboundp 'lsp-bridge-mode)
                 (not (bound-and-true-p lsp-bridge-mode)))
        (lsp-bridge-mode 1)))))

(defun mcg-zig-maybe-enable-lsp ()
  "Enable `lsp-bridge' for the current Zig buffer when prerequisites are ready."
  (cond
   ((not mcg-zig-enable-lsp)
    nil)
   ((not (mcg-zig--zls-available-p))
    (mcg-zig--warn-missing-zls))
   ((and (fboundp 'lsp-bridge-mode)
         (not (bound-and-true-p lsp-bridge-mode)))
    (lsp-bridge-mode 1))))

(defun mcg-zig-schedule-lsp ()
  "Enable Zig LSP after Emacs has been idle briefly."
  (mcg-zig--cancel-lsp-attach-timer)
  (cond
   ((not mcg-zig-enable-lsp)
    nil)
   ((not (mcg-zig--zls-available-p))
    (mcg-zig--warn-missing-zls))
   ((not (fboundp 'lsp-bridge-mode))
    nil)
   ((bound-and-true-p lsp-bridge-mode)
    t)
   (t
    (setq mcg-zig--lsp-attach-timer
          (run-with-idle-timer
           mcg-zig-lsp-idle-delay nil
           #'mcg-zig--enable-lsp-in-buffer (current-buffer))))))

(defun mcg-zig-schedule-lsp-for-open-buffers ()
  "Schedule deferred LSP startup for every open Zig buffer."
  (dolist (buffer (buffer-list))
    (when (buffer-live-p buffer)
      (with-current-buffer buffer
        (when (mcg-zig--buffer-p)
          (mcg-zig-schedule-lsp))))))

(defun mcg-zig--handle-module-load (category module)
  "Backfill Zig LSP startup after MODULE in CATEGORY loads."
  (when (and (eq category :tools)
             (eq module 'lsp-bridge))
    (mcg-zig-schedule-lsp-for-open-buffers)))

(add-hook 'mcg-module-load-hook #'mcg-zig--handle-module-load)
(with-eval-after-load 'lsp-bridge
  (mcg-zig-schedule-lsp-for-open-buffers))

;; File associations are managed by treesit-auto in init-treesit.el.
;; This module only configures mode behavior.
(add-hook 'zig-mode-hook #'mcg-zig-mode-setup)
(add-hook 'zig-ts-mode-hook #'mcg-zig-mode-setup)

;; zig-mode extension
(let ((zig-mode-path (mcg-extension-path "zig-mode")))
  (when zig-mode-path
    (add-to-list 'load-path zig-mode-path)))

(with-eval-after-load 'zig-mode
  (setq zig-format-on-save mcg-zig-format-on-save)
  (setq zig-format-show-buffer nil))
#+end_src

;;; ============================================================
;;; Zig Build Integration
;;; ============================================================

(defun mcg--zig-find-build-root ()
  "Find the root directory containing build.zig."
  (or (locate-dominating-file default-directory "build.zig")
      default-directory))

(defun mcg-zig--compile (command &optional root)
  "Run Zig compile COMMAND in ROOT through the unified compile layer."
  (let ((root (file-name-as-directory
               (or root (mcg--zig-find-build-root)))))
    (if (fboundp 'mcg-compile--run)
        (mcg-compile--run command root)
      (let ((default-directory root))
        (compile command)))))

(defun mcg-zig-build-command (action)
  "Return Zig build command for ACTION."
  (pcase action
    ('build "zig build")
    ('run "zig build run")
    ('test "zig build test")))

(defun mcg/zig-build ()
  "Run zig build."
  (interactive)
  (mcg-zig--compile (mcg-zig-build-command 'build)))

(defun mcg/zig-build-run ()
  "Run zig build run."
  (interactive)
  (mcg-zig--compile (mcg-zig-build-command 'run)))

(defun mcg/zig-build-test ()
  "Run zig build test."
  (interactive)
  (mcg-zig--compile (mcg-zig-build-command 'test)))

(defun mcg/zig-test ()
  "Run zig test on current file."
  (interactive)
  (let ((file (buffer-file-name)))
    (when file
      (save-buffer)
      (mcg-zig--compile (format "zig test %s" (shell-quote-argument file))))))

(defun mcg/zig-run ()
  "Run current Zig file."
  (interactive)
  (let ((file (buffer-file-name)))
    (when file
      (save-buffer)
      (mcg-zig--compile (format "zig run %s" (shell-quote-argument file))))))
#+end_src

;;; ============================================================
;;; Zig Keybindings

(defvar mcg-zig-mode-map
  (let ((map (make-sparse-keymap)))
    ;; Build
    (define-key map (kbd "C-c C-c b") #'mcg/zig-build)
    (define-key map (kbd "C-c C-c r") #'mcg/zig-build-run)
    (define-key map (kbd "C-c C-c t") #'mcg/zig-build-test)
    (define-key map (kbd "C-c C-c s") #'mcg/zig-build-step)
    (define-key map (kbd "C-c C-c R") #'mcg/zig-build-release)
    (define-key map (kbd "C-c C-c F") #'mcg/zig-build-release-fast)
    ;; Formatting
    (define-key map (kbd "C-c C-f") #'mcg-format-buffer)
    ;; Run/Test single file
    (define-key map (kbd "C-c C-r") #'mcg/zig-run)
    (define-key map (kbd "C-c C-t") #'mcg/zig-test)
    ;; Documentation
    (define-key map (kbd "C-c C-d d") #'mcg/zig-doc)
    (define-key map (kbd "C-c C-d s") #'mcg/zig-std-doc)
    map)
  "Keymap for Zig-specific commands.")

(defun mcg-zig-setup-keybindings ()
  "Setup Zig-specific keybindings."
  (use-local-map (make-composed-keymap mcg-zig-mode-map (current-local-map))))

;; Keybindings setup via hooks
(with-eval-after-load 'zig-mode
  (add-hook 'zig-mode-hook #'mcg-zig-setup-keybindings))
(when (mcg-treesit-available-p)
  (add-hook 'zig-ts-mode-hook #'mcg-zig-setup-keybindings))
#+end_src
