(defun mcg-python-mode-setup ()
  "Setup function for python-mode and python-ts-mode."
  (setq-local indent-tabs-mode nil)
  (setq-local tab-width 4)
  (setq-local python-indent-offset 4)
  (setq-local python-indent-guess-indent-offset nil)
  (subword-mode 1)
  (mcg-log "Python mode setup complete"))

(add-hook 'python-mode-hook #'mcg-python-mode-setup)
(add-hook 'python-ts-mode-hook #'mcg-python-mode-setup)

;;; ============================================================
;;; Helpers

(defun mcg-python--project-root ()
  "Return the current Python project root or `default-directory'."
  (mcg-project-root-or-default))

(defun mcg-python--save-and-compile (command &optional root)
  "Save buffers and run COMMAND with `compile' in ROOT."
  (save-some-buffers t)
  (let ((default-directory (or root default-directory)))
    (compile command)))

(defun mcg-python--current-file ()
  "Return the current buffer file or raise a user error."
  (or buffer-file-name
      (user-error "Current buffer is not visiting a file")))

(defun mcg-python--quoted-current-file ()
  "Return the shell-quoted current buffer file."
  (shell-quote-argument (mcg-python--current-file)))

(defun mcg-python--project-file-p (root file)
  "Return non-nil when ROOT contains FILE."
  (file-exists-p (expand-file-name file root)))

(defun mcg-python--project-tool (&optional root)
  "Return the detected Python project tool for ROOT, or nil.
Only return a tool when `mcg-python-detect-project-tools' is enabled and the
corresponding executable is available."
  (let ((root (file-name-as-directory (or root (mcg-python--project-root)))))
    (when mcg-python-detect-project-tools
      (cond
       ((and (mcg-python--project-file-p root "uv.lock")
             (executable-find "uv"))
        "uv")
       ((and (mcg-python--project-file-p root "poetry.lock")
             (executable-find "poetry"))
        "poetry")
       ((and (mcg-python--project-file-p root "pdm.lock")
             (executable-find "pdm"))
        "pdm")))))

(defun mcg-python--project-tool-command (tool args &optional root)
  "Return project-aware command parts for TOOL with ARGS in ROOT."
  (pcase (mcg-python--project-tool root)
    ("uv" (append (list "uv" "run" tool) args))
    ("poetry" (append (list "poetry" "run" tool) args))
    ("pdm" (append (list "pdm" "run" tool) args))
    (_ (cons tool args))))

(defun mcg-python--command-string (parts)
  "Return shell command string from command PARTS."
  (string-join parts " "))

(defun mcg-python--shell-command-string (parts)
  "Return a shell-quoted command string from command PARTS."
  (string-join (mapcar #'shell-quote-argument parts) " "))

(defun mcg-python--pytest-command (&optional root)
  "Return the best pytest command for ROOT."
  (let ((tool (mcg-python--project-tool root)))
    (if tool
        (mcg-python--command-string
         (mcg-python--project-tool-command "pytest" nil root))
      mcg-python-test-command)))

(defun mcg-python--pyproject-has-ruff-config-p (file)
  "Return non-nil when FILE contains a Ruff pyproject section."
  (and (file-exists-p file)
       (with-temp-buffer
         (insert-file-contents file)
         (goto-char (point-min))
         (re-search-forward "^\\[tool\\.ruff\\(?:\\.[^]]+\\)?\\]$" nil t))))

(defun mcg-python--ruff-project-config-p (&optional root)
  "Return non-nil when ROOT declares Ruff configuration."
  (let* ((root (file-name-as-directory (or root (mcg-python--project-root))))
         (pyproject (expand-file-name "pyproject.toml" root)))
    (or (mcg-python--project-file-p root ".ruff.toml")
        (mcg-python--project-file-p root "ruff.toml")
        (mcg-python--pyproject-has-ruff-config-p pyproject))))

(defun mcg-python--ruff-format-args (&optional root)
  "Return Ruff format args for ROOT without overriding project config."
  (if (mcg-python--ruff-project-config-p root)
      '("format")
    (list "format" "--config" (mcg-python--ruff-format-config-file))))

(defun mcg-python-format-candidates (&optional root)
  "Return project-aware Python formatter command candidates for ROOT."
  (let* ((root (file-name-as-directory (or root (mcg-python--project-root))))
         (project-tool (mcg-python--project-tool root))
         (ruff-format-args (mcg-python--ruff-format-args root))
         (project-candidate
          (when project-tool
            (mcg-python--project-tool-command
             "ruff" ruff-format-args root))))
    (append (when project-candidate (list project-candidate))
            (list (cons mcg-python-ruff-command ruff-format-args)
                  (list "black" "--skip-string-normalization")))))

(defun mcg-python--ruff-format-config-file ()
  "Return a Ruff config file path that forces single-quote formatting."
  (let* ((dir (or (and (boundp 'mcg-cache-dir) mcg-cache-dir)
                  temporary-file-directory))
         (file (expand-file-name mcg-python-ruff-format-config-file-name dir))
         (contents "[format]\nquote-style = \"single\"\n"))
    (unless (file-directory-p dir)
      (make-directory dir t))
    (unless (and (file-exists-p file)
                 (with-temp-buffer
                   (insert-file-contents file)
                   (equal (buffer-string) contents)))
      (with-temp-file file
        (insert contents)))
    file))

(defun mcg-python--ruff-format-command (&optional root)
  "Return Ruff format command parts for ROOT."
  (mcg-python--project-tool-command
   "ruff"
   (mcg-python--ruff-format-args root)
   root))

(defun mcg-python--ruff-command (&rest args)
  "Return the best Ruff command for ARGS, or raise a user error."
  (let ((candidate
         (cl-find-if (lambda (candidate)
                       (and (consp candidate)
                            (executable-find (car candidate))))
                     (let ((project-candidate
                            (mcg-python--project-tool-command "ruff" args)))
                       (append
                        (unless (equal (car project-candidate)
                                       mcg-python-ruff-command)
                          (list project-candidate))
                        (list (cons mcg-python-ruff-command args)))))))
    (unless candidate
      (user-error "No Ruff command found; install ruff or configure `mcg-python-ruff-command'"))
    (mcg-python--command-string candidate)))

(defun mcg/python-pytest-project ()
  "Run pytest for the current project."
  (interactive)
  (mcg-python--save-and-compile (mcg-python--pytest-command)
                                (mcg-python--project-root)))

(defun mcg/python-pytest-file ()
  "Run pytest for the current Python file."
  (interactive)
  (mcg-python--save-and-compile
   (format "%s %s" (mcg-python--pytest-command)
           (mcg-python--quoted-current-file))
   (mcg-python--project-root)))

(defun mcg/python-pytest-test-at-point ()
  "Run pytest for the current test symbol when available."
  (interactive)
  (let ((test-name (which-function)))
    (if (and test-name (not (string-empty-p test-name)))
        (mcg-python--save-and-compile
         (format "%s %s::%s"
                 (mcg-python--pytest-command)
                 (mcg-python--quoted-current-file)
                 (shell-quote-argument test-name))
         (mcg-python--project-root))
      (mcg/python-pytest-file))))
#+end_src

(defun mcg-python--repl-command ()
  "Return the command used by `mcg/python-repl'."
  (let* ((ipython (and mcg-python-use-ipython
                       (executable-find "ipython")))
         (interpreter (or ipython
                          (executable-find "python")
                          python-shell-interpreter))
         (args (if ipython mcg-python-ipython-args "")))
    (let ((python-shell-interpreter interpreter)
          (python-shell-interpreter-args args))
      (python-shell-calculate-command))))

(defun mcg/python-repl ()
  "Open a Python REPL, preferring IPython when available."
  (interactive)
  (run-python (mcg-python--repl-command) nil t))

(defvar mcg-python-mode-map
  (let ((map (make-sparse-keymap)))
    (define-key map (kbd "C-c C-c t") #'mcg/python-pytest-project)
    (define-key map (kbd "C-c C-c f") #'mcg/python-pytest-file)
    (define-key map (kbd "C-c C-c .") #'mcg/python-pytest-test-at-point)
    (define-key map (kbd "C-c C-c c") #'mcg/python-ruff-check)
    (define-key map (kbd "C-c C-c x") #'mcg/python-ruff-fix)
    (define-key map (kbd "C-c C-c F") #'mcg/python-ruff-format)
    (define-key map (kbd "C-c C-z") #'mcg/python-repl)
    map)
  "Keymap for Python-specific commands.")

(defun mcg-python-setup-keybindings ()
  "Setup Python-specific keybindings."
  (use-local-map (make-composed-keymap mcg-python-mode-map (current-local-map))))

(add-hook 'python-mode-hook #'mcg-python-setup-keybindings)
(add-hook 'python-ts-mode-hook #'mcg-python-setup-keybindings)
