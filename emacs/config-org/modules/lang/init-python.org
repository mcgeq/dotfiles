(defun mcg-python-mode-setup ()
  "Setup function for python-mode and python-ts-mode."
  (setq-local indent-tabs-mode nil)
  (setq-local tab-width 4)
  (setq-local python-indent-offset 4)
  (setq-local python-indent-guess-indent-offset nil)
  (subword-mode 1)
  (mcg-log "Python mode setup complete"))

(add-hook 'python-mode-hook #'mcg-python-mode-setup)
(add-hook 'python-ts-mode-hook #'mcg-python-mode-setup)

(defun mcg-python--save-and-compile (command &optional root)
  "Save buffers and run COMMAND with `compile' in ROOT."
  (save-some-buffers t)
  (let ((default-directory (or root default-directory)))
    (compile command)))

(defun mcg-python--current-file ()
  "Return the current buffer file or raise a user error."
  (or buffer-file-name
      (user-error "Current buffer is not visiting a file")))

(defun mcg-python--quoted-current-file ()
  "Return the shell-quoted current buffer file."
  (shell-quote-argument (mcg-python--current-file)))

(defun mcg-python--ruff-command (&rest args)
  "Return the best Ruff command for ARGS, or raise a user error."
  (let ((candidate
         (cl-find-if (lambda (candidate)
                       (and (consp candidate)
                            (executable-find (car candidate))))
                     (let ((project-candidate
                            (mcg-python--project-tool-command "ruff" args)))
                       (append
                        (unless (equal (car project-candidate)
                                       mcg-python-ruff-command)
                          (list project-candidate))
                        (list (cons mcg-python-ruff-command args)))))))
    (unless candidate
      (user-error "No Ruff command found; install ruff or configure `mcg-python-ruff-command'"))
    (mcg-python--command-string candidate)))
#+end_src

** Pytest Commands

#+begin_src emacs-lisp
;;; ============================================================
;;; Pytest Commands
;;; ============================================================

(defun mcg/python-pytest-project ()
  "Run pytest for the current project."
  (interactive)
  (mcg-python--save-and-compile (mcg-python--pytest-command)
                                (mcg-python--project-root)))

(defun mcg/python-pytest-file ()
  "Run pytest for the current Python file."
  (interactive)
  (mcg-python--save-and-compile
   (format "%s %s" (mcg-python--pytest-command)
           (mcg-python--quoted-current-file))
   (mcg-python--project-root)))

(defun mcg/python-pytest-test-at-point ()
  "Run pytest for the current test symbol when available."
  (interactive)
  (let ((test-name (which-function)))
    (if (and test-name (not (string-empty-p test-name)))
        (mcg-python--save-and-compile
         (format "%s %s::%s"
                 (mcg-python--pytest-command)
                 (mcg-python--quoted-current-file)
                 (shell-quote-argument test-name))
         (mcg-python--project-root))
      (mcg/python-pytest-file))))
#+end_src

(defun mcg-python--repl-command ()
  "Return the command used by `mcg/python-repl'."
  (let* ((ipython (and mcg-python-use-ipython
                       (executable-find "ipython")))
         (interpreter (or ipython
                          (executable-find "python")
                          python-shell-interpreter))
         (args (if ipython mcg-python-ipython-args "")))
    (let ((python-shell-interpreter interpreter)
          (python-shell-interpreter-args args))
      (python-shell-calculate-command))))

(defun mcg/python-repl ()
  "Open a Python REPL, preferring IPython when available."
  (interactive)
  (run-python (mcg-python--repl-command) nil t))
#+end_src

;;; ============================================================

(defvar mcg-python-mode-map
  (let ((map (make-sparse-keymap)))
    (define-key map (kbd "C-c C-c t") #'mcg/python-pytest-project)
    (define-key map (kbd "C-c C-c f") #'mcg/python-pytest-file)
    (define-key map (kbd "C-c C-c .") #'mcg/python-pytest-test-at-point)
    (define-key map (kbd "C-c C-c c") #'mcg/python-ruff-check)
    (define-key map (kbd "C-c C-c x") #'mcg/python-ruff-fix)
    (define-key map (kbd "C-c C-c F") #'mcg/python-ruff-format)
    (define-key map (kbd "C-c C-z") #'mcg/python-repl)
    map)
  "Keymap for Python-specific commands.")

(defun mcg-python-setup-keybindings ()
  "Setup Python-specific keybindings."
  (use-local-map (make-composed-keymap mcg-python-mode-map (current-local-map))))

(add-hook 'python-mode-hook #'mcg-python-setup-keybindings)
(add-hook 'python-ts-mode-hook #'mcg-python-setup-keybindings)
#+end_src
